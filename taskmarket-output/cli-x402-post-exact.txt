06611:   const keystore = await loadKeystore();

06613:   const r1 = await fetch(url, {
06614:     method: "POST",
06615:     redirect: "error",
06616:     headers: { "Content-Type": "application/json", ...legalHeaders },
06617:     body: JSON.stringify(body)
06618:   });
06619:   if (r1.status !== 402) {
06620:     if (r1.ok) {
06621:       return r1.json();
06622:     }
06623:     const text = await r1.text().catch(() => "");
06624:     throw new ApiError(r1.status, `POST ${path4} failed (${r1.status}): ${text}`);
06625:   }
06626:   const requirements = await r1.json();
06627:   if (!requirements.accepts || requirements.accepts.length === 0) {
06628:     throw new Error("No payment methods accepted by server");
06629:   }
06630:   const accept = requirements.accepts[0];
06631:   const { authorization, signature } = await createTransferAuthorization(accept, keystore);
06632:   const paymentPayload = {
06633:     x402Version: 2,
06634:     resource: requirements.resource,
06635:     accepted: {
06636:       scheme: accept.scheme,
06637:       network: accept.network,
06638:       amount: accept.amount,
06639:       asset: accept.asset,
06640:       payTo: accept.payTo,
06641:       maxTimeoutSeconds: accept.maxTimeoutSeconds,
06642:       extra: accept.extra
06643:     },
06644:     payload: { authorization, signature }
06645:   };
06646:   const r2 = await fetch(url, {
06647:     method: "POST",
06648:     redirect: "error",
06649:     headers: {
06650:       "Content-Type": "application/json",
06651:       ...legalHeaders,
06652:       "PAYMENT-SIGNATURE": Buffer.from(JSON.stringify(paymentPayload)).toString("base64")
06653:     },
06654:     body: JSON.stringify(body)

06660:       `POST ${path4} failed after payment (${r2.status}): ${JSON.stringify(result)}`
