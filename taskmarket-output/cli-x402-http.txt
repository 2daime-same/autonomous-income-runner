06170: import { createECDH, hkdfSync, createCipheriv as createCipheriv2, createDecipheriv as createDecipheriv2, randomBytes as randomBytes2 } from "crypto";
06171: var VERSION2 = 1;
06172: var EPH_PUB_LEN = 65;
06173: var IV_LEN = 12;
06174: var TAG_LEN = 16;
06175: var HEADER_LEN = 1 + EPH_PUB_LEN + IV_LEN + TAG_LEN;
06176: function ecdhSharedSecret(privateKeyHex, otherPubKeyHex) {
06177:   const ecdh = createECDH("secp256k1");
06178:   ecdh.setPrivateKey(Buffer.from(privateKeyHex.replace(/^0x/, ""), "hex"));
06179:   const shared = ecdh.computeSecret(Buffer.from(otherPubKeyHex, "hex"));
06180:   return shared;
06181: }
06182: function hkdfKey(sharedSecret) {
06183:   return Buffer.from(hkdfSync("sha256", sharedSecret, Buffer.alloc(0), "taskmarket-ecies-v1", 32));
06184: }
06185: function aesGcmEncrypt(key, iv, plaintext) {
06186:   const cipher = createCipheriv2("aes-256-gcm", key, iv);
06187:   const ciphertext = Buffer.concat([cipher.update(plaintext), cipher.final()]);
06188:   const tag = cipher.getAuthTag();
06189:   return { ciphertext, tag };
06190: }
06191: function aesGcmDecrypt(key, iv, tag, ciphertext) {
06192:   const decipher = createDecipheriv2("aes-256-gcm", key, iv);
06193:   decipher.setAuthTag(tag);
06194:   return Buffer.concat([decipher.update(ciphertext), decipher.final()]);
06195: }
06196: function derivePublicKey(privateKeyHex) {
06197:   const ecdh = createECDH("secp256k1");
06198:   ecdh.setPrivateKey(Buffer.from(privateKeyHex.replace(/^0x/, ""), "hex"));
06199:   return ecdh.getPublicKey(void 0, "uncompressed").toString("hex");
06200: }
06201: function deriveCompressedPublicKey(privateKeyHex) {
06202:   const ecdh = createECDH("secp256k1");
06203:   ecdh.setPrivateKey(Buffer.from(privateKeyHex.replace(/^0x/, ""), "hex"));
06204:   return ecdh.getPublicKey(void 0, "compressed").toString("hex");
06205: }
06206: function encryptForRecipient(fileBuffer, recipientPubKeyHex) {
06207:   const ephEcdh = createECDH("secp256k1");
06208:   ephEcdh.generateKeys();
06209:   const ephPubKey = ephEcdh.getPublicKey(void 0, "uncompressed");
06210:   const ephPrivKeyHex = ephEcdh.getPrivateKey("hex");
06211:   const shared = ecdhSharedSecret(ephPrivKeyHex, recipientPubKeyHex);
06212:   const aesKey = hkdfKey(shared);
06213:   const iv = randomBytes2(IV_LEN);
06214:   const { ciphertext, tag } = aesGcmEncrypt(aesKey, iv, fileBuffer);
06215:   return Buffer.concat([Buffer.from([VERSION2]), ephPubKey, iv, tag, ciphertext]);
06216: }
06217: function decryptWithPrivateKey(fileBuffer, privateKeyHex) {
06218:   if (fileBuffer.length < HEADER_LEN + 1) {
06219:     throw new Error("Decryption failed: invalid key or corrupted file");
06220:   }
06221:   const version2 = fileBuffer[0];
06222:   if (version2 !== VERSION2) {
06223:     throw new Error(`Decryption failed: unsupported file version 0x${version2.toString(16)}`);
06224:   }
06225:   const ephPubKeyHex = fileBuffer.subarray(1, 1 + EPH_PUB_LEN).toString("hex");
06226:   const iv = fileBuffer.subarray(1 + EPH_PUB_LEN, 1 + EPH_PUB_LEN + IV_LEN);
06227:   const tag = fileBuffer.subarray(1 + EPH_PUB_LEN + IV_LEN, 1 + EPH_PUB_LEN + IV_LEN + TAG_LEN);
06228:   const ciphertext = fileBuffer.subarray(1 + EPH_PUB_LEN + IV_LEN + TAG_LEN);
06229:   const shared = ecdhSharedSecret(privateKeyHex, ephPubKeyHex);
06230:   const aesKey = hkdfKey(shared);
06231:   try {
06232:     return aesGcmDecrypt(aesKey, iv, tag, ciphertext);
06233:   } catch {
06234:     throw new Error("Decryption failed: invalid key or corrupted file");
06235:   }
06236: }
